    Function

    An alien is trapped in a room where the door only opens after all platforms have been triggered and the alien collides with it.

    Create and play a game following the instructions below.

    How to play?

    Use the arrow keys to control the alien and make it jump onto each platform. Each time the alien lands on a platform, a new one appears! Keep going until the alien reaches the door!

    Instruction:

    1. Declear variables: let alien, crateimg, slabimg, doorimg, doorimg2
    2. Declear variables: floor1, floor2, crate1, crate2, crate3
    3. Declear variable: door
    4. Create and declear the following variable to false in the following sequence:
      1. platform1Created
      2. platform2Created
      3. platform3Created
      4. platform4Created
      5. platform5Created
    5. Create a preload function
      1. Set image '/assets/crate_E.png' to crateimg
      2. Set image '/assets/slabAngle_N.png' to slabimg
      3. Set image '/assets/doorCLosed_S.png' to doorimg
      4. Set image '/assets/doorOpen_S.png' to doorimg2
    6. Create a setup function
      1. Create a canvas
      2. Set gravity
    7. Create a floor1 sprite
      1. Position it at (200, 650)
      2. Set it 50px width and 50px height
      3. Set it to static
      4. Set its scale to 2
      5. Add the slab image to it
    8. Create a door sprite
      1. Position it at (150, 115)
      2. Size it to 25px width and 60px height
      3. Set it to static
      4. Set its scale x to -1.5 and scale y to 1.5
      5. Add the closed door image to it
    9. Create an alien sprite
      1. Position it at (200, 600)
      2. Size it to 50 x 80
      3. Set it to dynamic
      4. Add walk animation with 2 frames: 'alienYellow_walk1.png' and 'alienYellow_walk2.png'
      5. Add stand animation with 1 frame: 'alienYellow_front.png'
      6. Set its scale to 1
      7. Set its layer to 10
      8. Set its frame delay to 20
      9. Lock its rotation
    10. Create a draw function
      1. Set the background color to '#dbb691ff'
      2. Check if alien collides with floor1 and create the first slab platform if not already created
      3. Check if alien collides with floor1 and create the first crate platform if not already created
      4. Check if alien collides with crate1 and create the second crate platform if not already created
      5. Check if alien collides with crate2 and create the third crate platform if not already created
      6. Check if alien collides with crate3 and create the second slab platform if not already created
      7. Check if alien collides with the door and change the door image to open
    11. Create an update function
      1. Call the alienMovement function
    12. Create an alienMovement function
      1. Check if left arrow key is pressed
        1. Move the alien to the left by setting velocity to -2
        2. Change alien's animation to walk
        3. Set alien's scale to -1
      2. Check if right arrow key is pressed
        1. Move the alien to the right by setting velocity to 2
        2. Change alien's animation to walk
        3. Set alien's scale to 1
      3. If no arrow key is pressed
        1. Stop the alien's horizontal movement by setting velocity to 0
        2. Change alien's animation to stand
      4. Check if the up arrow key is pressed
        1. Make the alien jump by setting vertical velocity to -5
    13. Create a createSlab function
      1. Receive x, y coordinates as parameters
      2. Create a platform sprite at the given x, y position
      3. Set its scale to 2
      4. Set its layer to 2
      5. Assign the slab image to it
      6. Return the created platform sprite
    14. Create a createCrate function
      1. Receive x, y coordinates as parameters
      2. Create a platform sprite at the given x, y position
      3. Remove any existing colliders from the platform
      4. Add a new collider with offset (0, -15) and size (60, 1)
      5. Assign the crate image to it
      6. Set its scale to 2
      7. Set its layer to 2
      8. Return the created crate platform sprite
    
    let alien, crateimg, slabimg, doorimg, doorimg2;
    let floor1, floor2, crate1, crate2, crate3;
    let door;
    let platform1Created = false;
    let platform2Created = false;
    let platform3Created = false;
    let platform4Created = false;
    let platform5Created = false;
    
    function preload(){
      crateimg = loadImage('/assets/crate_E.png');
      slabimg = loadImage('/assets/slabAngle_N.png');
      doorimg = loadImage('/assets/doorCLosed_S.png');
      doorimg2 = loadImage('/assets/doorOpen_S.png');
    }
    
    function setup(){
      new Canvas(865, 750);
      world.gravity.y = 10;
    
      floor1 = new Sprite(200, 650, 50, 1, 'static');
      floor1.scale = 2;
      floor1.img = slabimg;
    
      door = new Sprite(150, 115, 25, 60, 'static');
      door.img = doorimg;
      door.layer = 3;
      door.scale.x = -1.5;
      door.scale.y = 1.5;
    
      alien = new Sprite(200, 600, 50, 80, 'dynamic');
      alien.addAni('walk', [
        '/assets/alienYellow_walk1.png',
        '/assets/alienYellow_walk2.png'
      ], 2);
      alien.addAni('stand', ['/assets/alienYellow_front.png'], 1);
      alien.scale = 1;
      alien.layer = 10;
      alien.frameDelay = 20;
      alien.rotationLock = true;
    }
    
    function draw(){
      background('#dbb691ff');
      if (alien.collides(floor1) && !platform1Created) {
        floor1 = createSlab(450, 620);
        platform1Created = true;
      }
    
      if (floor1 && alien.collides(floor1) && !platform2Created) {
        crate1 = createCrate(650, 500);
        platform2Created = true;
      }
    
      if (crate1 && alien.collides(crate1) && !platform3Created) {
        crate2 = createCrate(450, 400);
        platform3Created = true;
      }
    
      if (crate2 && alien.collides(crate2) && !platform4Created) {
        crate3 = createCrate(300, 300);
        platform4Created = true;
      }
    
      if (crate3 && alien.collides(crate3) && !platform5Created) {
        floor2 = createSlab(200, 200);
        platform5Created = true;
      }
    
      if (floor2 && alien.collides(door)) {
        door.img = doorimg2;
      }
    }
    
    function update(){
      alienMovement();
    }
    
    function alienMovement(){
      if (kb.pressing('left')){
        alien.vel.x = -2;
        alien.changeAni('walk');
        alien.scale.x = -1;
      } else if(kb.pressing('right')){
        alien.vel.x = 2;
        alien.changeAni('walk');
        alien.scale.x = 1;
      } else {
        alien.vel.x = 0;
        alien.changeAni('stand');
      }
    
      if (kb.presses('up')){
        alien.vel.y = -5;
      }
    }
    
    function createSlab(x, y) {
      let platform = new Sprite(x, y, 50, 1, 'static');
      platform.scale = 2;
      platform.layer = 2;
      platform.img = slabimg;
      return platform;
    }
    
    function createCrate(x, y) {
      let platform = new Sprite(x, y, 50, 1, 'static');
      platform.removeColliders();
      platform.addCollider(0, -15, 60, 1);
      platform.img = crateimg;
      platform.scale = 2;
      platform.layer = 2;
      return platform;
    }
